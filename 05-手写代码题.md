### 166. 实现防抖函数（debounce）

**回答：**

防抖：在事件被触发 n 秒后再执行回调，如果在这 n 秒内又被触发，则重新计时。

```javascript
function debounce(fn, delay) {
  let timer = null;
  return function(...args) {
    if (timer) clearTimeout(timer);
    timer = setTimeout(() => {
      fn.apply(this, args);
    }, delay);
  };
}
```

### 167. 实现节流函数（throttle）

**回答：**

节流：规定在一个单位时间内，只能触发一次函数。如果这个单位时间内触发多次函数，只有一次生效。

```javascript
// 定时器版本
function throttle(fn, delay) {
  let timer = null;
  return function(...args) {
    if (!timer) {
      timer = setTimeout(() => {
        fn.apply(this, args);
        timer = null;
      }, delay);
    }
  };
}

// 时间戳版本
function throttle(fn, delay) {
  let last = 0;
  return function(...args) {
    let now = Date.now();
    if (now - last > delay) {
      fn.apply(this, args);
      last = now;
    }
  };
}
```

### 168. 实现深拷贝（处理循环引用）

**回答：**

```javascript
function deepClone(obj, hash = new WeakMap()) {
  if (obj === null) return null;
  if (obj instanceof Date) return new Date(obj);
  if (obj instanceof RegExp) return new RegExp(obj);
  if (typeof obj !== 'object') return obj;

  if (hash.has(obj)) return hash.get(obj);

  const cloneObj = new obj.constructor();
  hash.set(obj, cloneObj);

  for (let key in obj) {
    if (obj.hasOwnProperty(key)) {
      cloneObj[key] = deepClone(obj[key], hash);
    }
  }
  return cloneObj;
}
```

### 175. 实现JSON.stringify和JSON.parse

**回答：**

```javascript
// 简易版 JSON.stringify
function jsonStringify(obj) {
  let type = typeof obj;
  if (type !== 'object') {
    if (/string|undefined|function/.test(type)) {
      obj = '"' + obj + '"';
    }
    return String(obj);
  } else {
    let json = [];
    let arr = Array.isArray(obj);
    for (let k in obj) {
      let v = obj[k];
      let type = typeof v;
      if (/string|undefined|function/.test(type)) {
        v = '"' + v + '"';
      } else if (type === 'object') {
        v = jsonStringify(v);
      }
      json.push((arr ? "" : '"' + k + '":') + String(v));
    }
    return (arr ? "[" : "{") + String(json) + (arr ? "]" : "}");
  }
}

// 简易版 JSON.parse
function jsonParse(jsonStr) {
  return eval("(" + jsonStr + ")"); // 或者使用 new Function('return ' + jsonStr)()
}
```

## 5.2 Promise相关（6道）

### 176. 手写Promise（基础版：resolve/reject/then）

**回答：**

```javascript
class MyPromise {
  constructor(executor) {
    this.state = 'pending';
    this.value = undefined;
    this.reason = undefined;
    this.onResolvedCallbacks = [];
    this.onRejectedCallbacks = [];

    const resolve = (value) => {
      if (this.state === 'pending') {
        this.state = 'fulfilled';
        this.value = value;
        this.onResolvedCallbacks.forEach(fn => fn());
      }
    };

    const reject = (reason) => {
      if (this.state === 'pending') {
        this.state = 'rejected';
        this.reason = reason;
        this.onRejectedCallbacks.forEach(fn => fn());
      }
    };

    try {
      executor(resolve, reject);
    } catch (err) {
      reject(err);
    }
  }

  then(onFulfilled, onRejected) {
    if (this.state === 'fulfilled') {
      onFulfilled(this.value);
    }
    if (this.state === 'rejected') {
      onRejected(this.reason);
    }
    if (this.state === 'pending') {
      this.onResolvedCallbacks.push(() => onFulfilled(this.value));
      this.onRejectedCallbacks.push(() => onRejected(this.reason));
    }
  }
}
```

```javascript
Promise.myAny = function(promises) {
  return new Promise((resolve, reject) => {
    let count = 0;
    let errors = [];
    const len = promises.length;
    if (len === 0) return reject(new AggregateError([], 'All promises were rejected'));

    promises.forEach((p, i) => {
      Promise.resolve(p).then(res => {
        resolve(res);
      }, err => {
        count++;
        errors[i] = err;
        if (count === len) reject(new AggregateError(errors, 'All promises were rejected'));
      });
    });
  });
};
```

### 181. 实现串行和并行限制的异步调度器

**回答：**

```javascript
class Scheduler {
  constructor(limit) {
    this.limit = limit;
    this.count = 0;
    this.queue = [];
  }

  add(fn) {
    return new Promise((resolve) => {
      this.queue.push({ fn, resolve });
      this.run();
    });
  }

  run() {
    if (this.count < this.limit && this.queue.length) {
      this.count++;
      const { fn, resolve } = this.queue.shift();
      fn().then(() => {
        resolve();
        this.count--;
        this.run();
      });
    }
  }
}
```

```javascript
// 递归实现
function flatten(arr) {
  let result = [];
  for (let item of arr) {
    if (Array.isArray(item)) {
      result = result.concat(flatten(item));
    } else {
      result.push(item);
    }
  }
  return result;
}

// 指定层数实现
function flat(arr, depth = 1) {
  return depth > 0 
    ? arr.reduce((acc, val) => acc.concat(Array.isArray(val) ? flat(val, depth - 1) : val), [])
    : arr.slice();
}
```

### 185. 实现洗牌算法（Shuffle）

**回答：**

```javascript
function shuffle(arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    let j = Math.floor(Math.random() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}
```

### 187. 实现发布订阅模式（EventEmitter）

**回答：**

```javascript
class EventEmitter {
  constructor() {
    this.events = {};
  }
  on(name, fn) {
    if (!this.events[name]) this.events[name] = [];
    this.events[name].push(fn);
  }
  off(name, fn) {
    if (this.events[name]) {
      this.events[name] = this.events[name].filter(f => f !== fn);
    }
  }
  emit(name, ...args) {
    if (this.events[name]) {
      this.events[name].forEach(fn => fn(...args));
    }
  }
  once(name, fn) {
    const wrapper = (...args) => {
      fn(...args);
      this.off(name, wrapper);
    };
    this.on(name, wrapper);
  }
}
```

```javascript
// Vue2 (Object.defineProperty)
function defineReactive(obj, key, val) {
  Object.defineProperty(obj, key, {
    get() {
      console.log('get', key);
      return val;
    },
    set(newVal) {
      if (newVal !== val) {
        console.log('set', key);
        val = newVal;
      }
    }
  });
}

// Vue3 (Proxy)
function reactive(obj) {
```

### 190. 实现一个简易版的Vuex（Store类）

**回答：**

```javascript
class Store {
  constructor(options) {
    this._state = reactive(options.state); // 使用 Vue3 响应式
    this._mutations = options.mutations;
    this._actions = options.actions;
  }

  get state() {
    return this._state;
  }

  commit = (type, payload) => {
    this._mutations[type](this._state, payload);
  }

  dispatch = (type, payload) => {
    this._actions[type](this, payload);
  }
}
```
